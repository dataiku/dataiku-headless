# Solution

```bash
dku insight create "Revenue by Region" --type chart --dataset orders -P {project}
dku insight create "Orders Table" --type dataset_table --dataset orders -P {project}
dku insight list -P {project} -o json | jq -r '.[] | select(.name=="Revenue by Region") | .id' > /tmp/{project}_dashboard_chart_id.txt
dku insight list -P {project} -o json | jq -r '.[] | select(.name=="Orders Table") | .id' > /tmp/{project}_dashboard_table_id.txt
python3 -c 'import json,pathlib; iid=pathlib.Path("/tmp/{project}_dashboard_chart_id.txt").read_text().strip(); payload={"id":iid,"projectKey":"{project}","type":"chart","name":"Revenue by Region","listed":True,"params":{"engineType":"LINO","datasetSmartName":"orders","def":{"type":"grouped_columns","variant":"normal","name":"Revenue by Region","userEditedName":True,"genericDimension0":[{"column":"region","type":"ALPHANUM","isA":"dimension","maxValues":100,"filters":[],"sort":{"type":"NATURAL","sortAscending":True},"numParams":{"mode":"FIXED_NB","emptyBinsMode":"ZEROS"}}],"genericDimension1":[],"genericMeasures":[{"column":"amount","function":"SUM","type":"NUMERICAL","displayed":True,"isA":"measure","displayAxis":"axis1","displayType":"column","computeMode":"NORMAL"}],"facetDimension":[],"animationDimension":[],"filters":[],"xDimension":[],"yDimension":[],"tooltipMeasures":[],"showLegend":True,"colorOptions":{"singleColor":"#2678b2","transparency":0.75}},"refreshableSelection":{"selection":{"samplingMethod":"FULL","maxRecords":10000},"autoRefreshSample":False}}}; open("/tmp/{project}_dashboard_chart.json","w").write(json.dumps(payload))'
dku insight set-definition "$(cat /tmp/{project}_dashboard_chart_id.txt)" -d @/tmp/{project}_dashboard_chart.json -P {project}
dku dashboard create "Operations Dashboard" -P {project}
dku dashboard list -P {project} -o json | jq -r '.[] | select(.name=="Operations Dashboard") | .id' > /tmp/{project}_dashboard_id.txt
python3 -c 'import json,pathlib; dash=pathlib.Path("/tmp/{project}_dashboard_id.txt").read_text().strip(); chart=pathlib.Path("/tmp/{project}_dashboard_chart_id.txt").read_text().strip(); table=pathlib.Path("/tmp/{project}_dashboard_table_id.txt").read_text().strip(); payload={"projectKey":"{project}","id":dash,"name":"Operations Dashboard","pages":[{"id":"page1","title":"Overview","displayedTitle":"Overview","show":True,"showTitle":False,"enableCrossFilters":True,"backgroundColor":"#FFFEF9","grid":{"tiles":[{"tileType":"INSIGHT","insightId":chart,"insightType":"chart","displayMode":"INSIGHT","box":{"top":0,"left":0,"width":18,"height":14},"autoLoad":True,"titleOptions":{"showTitle":"YES","title":"Revenue by Region","displayedTitle":"Revenue by Region"}},{"tileType":"INSIGHT","insightId":table,"insightType":"dataset_table","displayMode":"INSIGHT","box":{"top":0,"left":18,"width":18,"height":14},"autoLoad":True,"titleOptions":{"showTitle":"YES","title":"Orders Table","displayedTitle":"Orders Table"}}]}}]}; open("/tmp/{project}_dashboard.json","w").write(json.dumps(payload))'
dku dashboard set-definition "$(cat /tmp/{project}_dashboard_id.txt)" -d @/tmp/{project}_dashboard.json -P {project}
```
